/* FILE: cookie.js */

function login(event) {
  event.preventDefault();
  let user = document.getElementById("username").value.trim();
  if (!user) {
    alert("Please enter a username.");
    return;
  }
  document.cookie = "player=" + encodeURIComponent(user) + "; path=/; SameSite=Lax";
  window.location.href = "menu.html";
}

function checkLogin() {
  if (!getUsername()) {
    alert("Please login first!");
    window.location.href = "index.html";
  }
}

function logout() {
  document.cookie = "player=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;";
  window.location.href = "index.html";
}

// NEW FUNCTION TO FETCH THE USERNAME
function getUsername() {
  const cookies = document.cookie.split(';'); // Split all cookies into an array
  for (let i = 0; i < cookies.length; i++) {
    let cookie = cookies[i].trim(); // Remove whitespace
    // Check if this cookie string begins with "player="
    if (cookie.startsWith('player=')) {
      // If it does, return the value after "player="
      return decodeURIComponent(cookie.substring('player='.length, cookie.length));
    }
  }
  // Return null if the cookie wasn't found
  return null;
}

/* FILE: game.js */
// Manages level progression and progress bar

const levelOrder = [
  'soceng.html', 'phar.html', 'whl.html', 'wesp.html', 'sph.html',
  'htph.html', 'emph.html', 'anph.html', 'clph.html', 'viph.html',
  'whph.html', 'puph.html', 'dcph.html', 'etph.html', 'seph.html',
  'iph.html', 'smh.html', 'dmph.html', 'mitm.html', 'smph.html'
];

const COMPLETED_LEVELS_KEY = 'phishingSimCompletedLevels';
const LAST_PLAYED_INDEX_KEY = 'phishingSimLastPlayedIndex';

function getCurrentLevelIndex() {
  const currentPage = window.location.pathname.split("/").pop();
  return levelOrder.indexOf(currentPage);
}

function getCompletedLevels() {
  try {
    const raw = localStorage.getItem(COMPLETED_LEVELS_KEY);
    if (!raw) {
      return [];
    }
    const parsed = JSON.parse(raw);
    if (!Array.isArray(parsed)) {
      return [];
    }
    return parsed
      .filter((value) => Number.isInteger(value) && value >= 0 && value < levelOrder.length)
      .sort((a, b) => a - b);
  } catch (error) {
    return [];
  }
}

function saveCompletedLevels(completedIndexes) {
  const uniqueSorted = [...new Set(completedIndexes)].sort((a, b) => a - b);
  localStorage.setItem(COMPLETED_LEVELS_KEY, JSON.stringify(uniqueSorted));
}

function markLevelCompleted(levelIndex) {
  if (!Number.isInteger(levelIndex) || levelIndex < 0) {
    return;
  }
  const completed = getCompletedLevels();
  if (!completed.includes(levelIndex)) {
    completed.push(levelIndex);
    saveCompletedLevels(completed);
  }
}

function setLastPlayedIndex(levelIndex) {
  if (!Number.isInteger(levelIndex) || levelIndex < 0) {
    return;
  }
  localStorage.setItem(LAST_PLAYED_INDEX_KEY, String(levelIndex));
}

function getLastPlayedIndex() {
  const raw = localStorage.getItem(LAST_PLAYED_INDEX_KEY);
  const parsed = Number(raw);
  if (!Number.isInteger(parsed) || parsed < 0 || parsed >= levelOrder.length) {
    return 0;
  }
  return parsed;
}

function getFirstUnfinishedLevelIndex() {
  const completed = new Set(getCompletedLevels());
  for (let i = 0; i < levelOrder.length; i += 1) {
    if (!completed.has(i)) {
      return i;
    }
  }
  return levelOrder.length - 1;
}

function proceedToNextLevel() {
  const currentIndex = getCurrentLevelIndex();
  if (currentIndex !== -1) {
    markLevelCompleted(currentIndex);
  }

  if (currentIndex !== -1 && currentIndex < levelOrder.length - 1) {
    const nextLevel = levelOrder[currentIndex + 1];
    setLastPlayedIndex(currentIndex + 1);
    window.location.href = nextLevel;
  } else {
    setLastPlayedIndex(levelOrder.length - 1);
    window.location.href = 'end.html';
  }
}

function updateProgressBar() {
  const currentIndex = getCurrentLevelIndex();
  const completedCount = getCompletedLevels().length;
  if (currentIndex !== -1) {
    setLastPlayedIndex(currentIndex);
    const progressed = Math.max(currentIndex + 1, completedCount);
    const progressPercent = (progressed / levelOrder.length) * 100;
    const progressBar = document.getElementById('progress-bar') || document.getElementById('Progress-bar');
    if(progressBar) {
        progressBar.style.width = progressPercent + '%';
    }
  }
}

function updateProgresBar() {
  updateProgressBar();
}

function updateprogressbar() {
  updateProgressBar();
}

/* Replace the existing displayUsername function in game.js with this one */

function displayUsername() {
    const user = getUsername(); // Calls the function from cookie.js
    if (user) {
        // Update the main welcome message if it exists on the page
        const welcomeMessage = document.getElementById('welcome-message');
        if (welcomeMessage) {
            welcomeMessage.textContent = `Welcome, ${user}!`;
        }

        // Update the in-level player name placeholder if it exists
        const playerNameSpan = document.getElementById('playerName');
        if (playerNameSpan) {
            playerNameSpan.textContent = user;
        }
    }
}

function renderMenuProgress() {
  const menuLinks = document.querySelectorAll('a.menu-link');
  if (!menuLinks.length) {
    return;
  }

  const completed = new Set(getCompletedLevels());

  menuLinks.forEach((link) => {
    const href = link.getAttribute('href');
    const levelIndex = levelOrder.indexOf(href);
    if (levelIndex === -1) {
      return;
    }

    if (link.querySelector('.status-badge')) {
      return;
    }

    const badge = document.createElement('span');
    badge.className = 'status-badge';
    if (completed.has(levelIndex)) {
      badge.classList.add('status-complete');
      badge.textContent = 'Completed';
    } else {
      badge.classList.add('status-pending');
      badge.textContent = 'Pending';
    }
    link.appendChild(document.createTextNode(' '));
    link.appendChild(badge);
  });

  const summary = document.getElementById('menu-summary');
  if (summary) {
    summary.textContent = `Progress: ${completed.size}/${levelOrder.length} levels completed`;
  }

  const resumeLink = document.getElementById('resume-link');
  if (resumeLink) {
    const firstUnfinishedIndex = getFirstUnfinishedLevelIndex();
    const lastPlayedIndex = getLastPlayedIndex();
    const resumeIndex = completed.has(lastPlayedIndex) ? firstUnfinishedIndex : lastPlayedIndex;
    const allDone = completed.size === levelOrder.length;
    resumeLink.href = allDone ? 'end.html' : levelOrder[resumeIndex];
    resumeLink.textContent = allDone ? 'View Completion Page' : 'Resume Training';
  }
}

function renderEndSummary() {
  const completedCount = getCompletedLevels().length;
  const summary = document.getElementById('completion-summary');
  if (!summary) {
    return;
  }
  if (completedCount >= levelOrder.length) {
    summary.textContent = `Excellent work. You completed all ${levelOrder.length} levels.`;
  } else {
    summary.textContent = `You completed ${completedCount} of ${levelOrder.length} levels. Continue from the menu to finish the simulator.`;
  }
}

function resetGameProgress() {
  localStorage.removeItem(COMPLETED_LEVELS_KEY);
  localStorage.removeItem(LAST_PLAYED_INDEX_KEY);
  window.location.href = 'menu.html';
}

function confirmResetGameProgress() {
  const shouldReset = window.confirm('Reset your saved training progress?');
  if (shouldReset) {
    resetGameProgress();
  }
}

===== server.js =====
const express = require("express");
const path = require("path");
const fs = require("fs");
const crypto = require("crypto");

const app = express();
const PORT = process.env.PORT || 3000;
const DATA_DIR = path.join(__dirname, "data");
const DATA_FILE = path.join(DATA_DIR, "users.json");
const PUBLIC_DIR = path.join(__dirname, "public");

app.use(express.json({ limit: "1mb" }));
app.use(express.static(PUBLIC_DIR));

function ensureDataFile() {
  if (!fs.existsSync(DATA_DIR)) {
    fs.mkdirSync(DATA_DIR, { recursive: true });
  }
  if (!fs.existsSync(DATA_FILE)) {
    fs.writeFileSync(DATA_FILE, JSON.stringify({ users: {}, nameIndex: {} }, null, 2));
  }
}

function loadData() {
  ensureDataFile();
  try {
    const raw = fs.readFileSync(DATA_FILE, "utf8");
    const parsed = JSON.parse(raw);
    if (!parsed.users || !parsed.nameIndex) {
      return { users: {}, nameIndex: {} };
    }
    return parsed;
  } catch (error) {
    return { users: {}, nameIndex: {} };
  }
}

function saveData(data) {
  ensureDataFile();
  fs.writeFileSync(DATA_FILE, JSON.stringify(data, null, 2));
}

function normalizeName(name) {
  return String(name || "").trim().toLowerCase();
}

function getUserIdFromRequest(req) {
  return req.header("x-user-id") || "";
}

app.get("/api/health", (req, res) => {
  res.json({ ok: true });
});

app.post("/api/login", (req, res) => {
  const name = (req.body && req.body.name) ? String(req.body.name).trim() : "";
  if (!name) {
    res.status(400).json({ error: "Name required" });
    return;
  }

  const data = loadData();
  const key = normalizeName(name);
  let userId = data.nameIndex[key];

  if (!userId) {
    userId = `u_${crypto.randomUUID()}`;
    data.nameIndex[key] = userId;
    data.users[userId] = {
      id: userId,
      name,
      state: null,
      updatedAt: new Date().toISOString()
    };
    saveData(data);
  }

  const user = data.users[userId];
  res.json({ userId, state: user ? user.state : null });
});

app.get("/api/state", (req, res) => {
  const userId = getUserIdFromRequest(req);
  if (!userId) {
    res.status(401).json({ error: "Missing user id" });
    return;
  }
  const data = loadData();
  const user = data.users[userId];
  if (!user) {
    res.status(404).json({ error: "User not found" });
    return;
  }
  res.json(user.state || null);
});

app.post("/api/state", (req, res) => {
  const userId = getUserIdFromRequest(req);
  if (!userId) {
    res.status(401).json({ error: "Missing user id" });
    return;
  }
  const data = loadData();
  const user = data.users[userId];
  if (!user) {
    res.status(404).json({ error: "User not found" });
    return;
  }
  user.state = req.body || null;
  user.updatedAt = new Date().toISOString();
  data.users[userId] = user;
  saveData(data);
  res.json({ ok: true });
});

app.get("*", (req, res) => {
  res.sendFile(path.join(PUBLIC_DIR, "index.html"));
});

app.listen(PORT, () => {
  console.log(`AegisPhish Lab running on http://localhost:${PORT}`);
});

  function getDefaultState() {
    return {

  function loadState() {
    const fallback = getDefaultState();
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) {
        return fallback;
      }
      const parsed = JSON.parse(raw);
      return {
        ...fallback,
        ...parsed,
        settings: { ...fallback.settings, ...(parsed.settings || {}) },
        stats: { ...fallback.stats, ...(parsed.stats || {}) },
        achievements: Array.isArray(parsed.achievements) ? parsed.achievements : [],
        history: Array.isArray(parsed.history) ? parsed.history : []
      };
    } catch (error) {
      return fallback;
    }
  }

    localStorage.setItem(STORAGE_KEY, JSON.stringify(state));

    el.timePerQuestion.addEventListener("change", updateSettingsFromForm);
    el.libraryBody.addEventListener("click", onLibraryAction);
    window.addEventListener("keydown", onKeyControls);
  }

  async function onLogin(event) {
    event.preventDefault();
    const name = (el.usernameInput.value || "").trim();
    if (!name) {
      window.alert("Please enter your analyst name.");
      return;
    }
    if (apiAvailable) {
      const result = await apiLogin(name);
      if (result && result.userId) {
        setUserId(result.userId);
        if (result.state) {
          state = loadImportedState(result.state);
        }
        state.profile = { name };
        saveState();
        syncUI();
        return;
      }
    }
    state.profile = { name };
    saveState();
    syncUI();
  }

  function onLogout() {
    clearTimer();
    state.profile = null;
    state.session = null;
    clearUserId();
    saveState();
    syncUI();
  }

  function onContinueSession() {
    if (!state.session) {
      window.alert("No active session found.");
      return;
    }
    showSession();
    renderScenario();
  }

  function updateSettingsFromForm() {
    const seconds = Number(el.timePerQuestion.value);
    state.settings.timerEnabled = Boolean(el.timerToggle.checked);
    state.settings.timePerQuestion = Number.isFinite(seconds) ? clamp(seconds, 10, 120) : 35;
    el.timePerQuestion.value = String(state.settings.timePerQuestion);
    saveState();
  }

  function startSession(mode, singleScenarioId) {
    const queue = buildQueue(mode, singleScenarioId);
    if (!queue.length) {
      window.alert("No scenarios available.");
      return;
    }

    clearTimer();
    state.session = {
      mode,
      queue,
      currentIndex: 0,
      answered: {},
      startedAt: Date.now(),
      currentTimer: state.settings.timePerQuestion
    };

    saveState();
    showSession();
    renderScenario();
  }

  function buildQueue(mode, singleScenarioId) {
    if (singleScenarioId) {
      return [singleScenarioId];
    }
    if (mode === "challenge") {
      return shuffle(SCENARIOS.map((s) => s.id)).slice(0, 10);
    }
    return SCENARIOS.map((s) => s.id);
  }

  function showSession() {
    el.authView.classList.add("hidden");
    el.dashboardView.classList.add("hidden");
    el.sessionView.classList.remove("hidden");
    el.logoutBtn.classList.remove("hidden");
  }

  function showDashboard() {
    el.authView.classList.add("hidden");
    el.dashboardView.classList.remove("hidden");
    el.sessionView.classList.add("hidden");
    el.logoutBtn.classList.remove("hidden");
  }

  function renderScenario() {
    if (!state.session) {
      syncUI();
      return;
    }
    const session = state.session;
    const scenarioId = session.queue[session.currentIndex];
    const scenario = getScenarioById(scenarioId);
    if (!scenario) {
      endSession();
      return;
    }

    el.sessionMode.textContent = session.mode === "challenge" ? "Rapid Challenge" : "Guided Learning";
    el.scenarioTitle.textContent = scenario.title;
    el.scenarioPrompt.textContent = scenario.prompt;
    el.scenarioMessage.textContent = scenario.message;
    el.sessionCounter.textContent = `Scenario ${session.currentIndex + 1}/${session.queue.length}`;
    el.sessionProgressBar.style.width = `${((session.currentIndex + 1) / session.queue.length) * 100}%`;
    el.optionContainer.innerHTML = "";
    el.feedbackPanel.classList.add("hidden");
    el.nextBtn.classList.add("hidden");

    scenario.options.forEach((option, idx) => {
      const btn = document.createElement("button");
      btn.type = "button";
      btn.className = "choice-btn";
      btn.textContent = `${idx + 1}. ${option.text}`;
      btn.dataset.optionIndex = String(idx);
      btn.addEventListener("click", () => handleAnswer(idx));
      el.optionContainer.appendChild(btn);
    });

    applyTimerRules();
    saveState();
  }

  function applyTimerRules() {
    clearTimer();
    const useTimer = state.settings.timerEnabled || state.session.mode === "challenge";
    if (!useTimer) {
      el.timerBox.classList.add("hidden");
      return;
    }
    el.timerBox.classList.remove("hidden");
    state.session.currentTimer = state.settings.timePerQuestion;
    el.timerValue.textContent = String(state.session.currentTimer);
    timerHandle = window.setInterval(() => {
      if (!state.session) {
        clearTimer();
        return;
      }
      state.session.currentTimer -= 1;
      el.timerValue.textContent = String(state.session.currentTimer);
      if (state.session.currentTimer <= 0) {
        clearTimer();
        forceTimeoutAnswer();
      }
    }, 1000);
  }

  function forceTimeoutAnswer() {
    const scenario = getCurrentScenario();
    if (!scenario) {
      return;
    }
    const wrongIndex = scenario.options.findIndex((opt) => !opt.safe);
    const index = wrongIndex >= 0 ? wrongIndex : 0;
    handleAnswer(index, true);
  }

  function handleAnswer(optionIndex, timedOut) {
    const scenario = getCurrentScenario();
    if (!scenario || !state.session) {
      return;
    }
    const answerKey = `${state.session.currentIndex}`;
    if (state.session.answered[answerKey]) {
      return;
    }

    clearTimer();
    const option = scenario.options[optionIndex];
    const isCorrect = Boolean(option && option.safe);
    const points = isCorrect ? (100 + scenario.difficulty * 25) : -20;
    state.stats.answered += 1;
    state.stats.score = Math.max(0, state.stats.score + points);
    state.stats.streak = isCorrect ? state.stats.streak + 1 : 0;
    state.stats.bestStreak = Math.max(state.stats.bestStreak, state.stats.streak);
    if (isCorrect) {
      state.stats.correct += 1;
      if (!state.stats.completedIds.includes(scenario.id)) {
        state.stats.completedIds.push(scenario.id);
      }
    }

    state.history.unshift({
      timestamp: new Date().toISOString(),
      scenarioId: scenario.id,
      result: isCorrect ? "correct" : "incorrect",
      mode: state.session.mode,
      timedOut: Boolean(timedOut)
    });
    state.history = state.history.slice(0, 100);

    state.session.answered[answerKey] = true;
    el.feedbackPanel.classList.remove("hidden");
    el.feedbackPanel.classList.toggle("ok", isCorrect);
    el.feedbackPanel.classList.toggle("bad", !isCorrect);
    el.feedbackTitle.textContent = isCorrect ? "Correct Decision" : "Risky Decision";
    el.feedbackBody.textContent = timedOut
      ? `Time expired. ${option.explanation}`
      : option.explanation;

    const choiceButtons = el.optionContainer.querySelectorAll(".choice-btn");
    choiceButtons.forEach((button, idx) => {
      button.disabled = true;
      if (idx === optionIndex) {
        button.classList.add(isCorrect ? "choice-correct" : "choice-wrong");
      }
    });

    updateAchievements();
    saveState();
    el.nextBtn.classList.remove("hidden");
    refreshDashboardData();
  }

  function moveToNextScenario() {
    if (!state.session) {
      return;
    }
    state.session.currentIndex += 1;
    if (state.session.currentIndex >= state.session.queue.length) {
      endSession();
      return;
    }
    renderScenario();
  }

  function endSessionEarly() {
    if (!state.session) {
      return;
    }
    const confirmed = window.confirm("End this session and return to dashboard?");
    if (!confirmed) {
      return;
    }
    endSession();
  }

  function endSession() {
    clearTimer();
    state.session = null;
    saveState();
    syncUI();
  }

  function resetProgress() {
    const confirmed = window.confirm("This will erase your score, achievements, and history. Continue?");
    if (!confirmed) {
      return;
    }
    const name = state.profile ? state.profile.name : null;
    const fresh = getDefaultState();
    if (name) {
      fresh.profile = { name };
    }
    Object.assign(state, fresh);
    saveState();
    syncUI();
  }

  function exportProgress() {
    const payload = JSON.stringify(state, null, 2);
    const blob = new Blob([payload], { type: "application/json" });
    const url = URL.createObjectURL(blob);
    const anchor = document.createElement("a");
    anchor.href = url;
    anchor.download = "aegisphish-progress.json";
    anchor.click();
    URL.revokeObjectURL(url);
  }

  function importProgress(event) {
    const file = event.target.files && event.target.files[0];
    if (!file) {
      return;
    }
    const reader = new FileReader();
    reader.onload = () => {
      try {
        const imported = JSON.parse(String(reader.result || "{}"));
        const safe = loadImportedState(imported);
        Object.assign(state, safe);
        saveState();
        syncUI();
        window.alert("Progress imported.");
      } catch (error) {
        window.alert("Import failed. Invalid JSON format.");
      } finally {
        el.importInput.value = "";
      }
    };
    reader.readAsText(file);
  }

  function loadImportedState(imported) {
    const fallback = getDefaultState();
    return {
      ...fallback,
      ...imported,
      settings: { ...fallback.settings, ...(imported.settings || {}) },
      stats: {
        ...fallback.stats,
        ...(imported.stats || {}),
        completedIds: Array.isArray(imported.stats && imported.stats.completedIds)
          ? imported.stats.completedIds.filter((id) => SCENARIOS.some((s) => s.id === id))
          : []
      },
      achievements: Array.isArray(imported.achievements) ? imported.achievements : [],
      history: Array.isArray(imported.history) ? imported.history.slice(0, 100) : [],
      session: null
    };
  }

  function onLibraryAction(event) {
    const target = event.target;
    if (!(target instanceof HTMLButtonElement)) {
      return;
    }
    const id = target.dataset.practiceId;
    if (!id) {
      return;
    }
    startSession("practice", id);
  }

  function onKeyControls(event) {
    if (!state.session || el.sessionView.classList.contains("hidden")) {
      return;
    }
    if (event.key === "1" || event.key === "2") {
      const idx = Number(event.key) - 1;
      handleAnswer(idx, false);
    }
    if (event.key.toLowerCase() === "n" && !el.nextBtn.classList.contains("hidden")) {
      moveToNextScenario();
    }
  }

  function updateAchievements() {
    ACHIEVEMENTS.forEach((achievement) => {
      const unlocked = state.achievements.includes(achievement.id);
      if (!unlocked && achievement.check(state)) {
        state.achievements.push(achievement.id);
      }
    });
  }

  function renderAchievements() {
    el.achievementList.innerHTML = "";
    ACHIEVEMENTS.forEach((achievement) => {
      const item = el.achievementTemplate.content.firstElementChild.cloneNode(true);
      const unlocked = state.achievements.includes(achievement.id);
      item.classList.toggle("locked", !unlocked);
      item.querySelector("h4").textContent = achievement.title;
      item.querySelector("p").textContent = achievement.description;
      el.achievementList.appendChild(item);
    });
  }

  function renderLibrary() {
    el.libraryBody.innerHTML = "";
    const completed = new Set(state.stats.completedIds);
    SCENARIOS.forEach((scenario) => {
      const row = document.createElement("tr");
      row.innerHTML = [
        `<td>${escapeHtml(scenario.title)}</td>`,
        `<td>${escapeHtml(scenario.type)}</td>`,
        `<td>${scenario.difficulty}</td>`,
        `<td>${completed.has(scenario.id) ? "Completed" : "Pending"}</td>`,
        `<td><button type="button" data-practice-id="${scenario.id}">Practice</button></td>`
      ].join("");
      el.libraryBody.appendChild(row);
    });
  }

  function refreshDashboardData() {
    const answered = state.stats.answered;
    const correct = state.stats.correct;
    const accuracy = answered > 0 ? Math.round((correct / answered) * 100) : 0;
    el.welcomeText.textContent = `Welcome, ${state.profile ? state.profile.name : "Analyst"}`;
    el.summaryText.textContent = "Analyze, decide, and build resilience against phishing campaigns.";
    el.scoreStat.textContent = String(state.stats.score);
    el.accuracyStat.textContent = `${accuracy}%`;
    el.streakStat.textContent = String(state.stats.bestStreak);
    el.completedStat.textContent = `${state.stats.completedIds.length}/${SCENARIOS.length}`;
    el.continueHint.textContent = state.session
      ? `${state.session.mode} - scenario ${state.session.currentIndex + 1}/${state.session.queue.length}`
      : "No active session";
    el.timerToggle.checked = Boolean(state.settings.timerEnabled);
    el.timePerQuestion.value = String(state.settings.timePerQuestion);
  }

  function syncUI() {
    clearTimer();
    if (!state.profile) {
      el.authView.classList.remove("hidden");
      el.dashboardView.classList.add("hidden");
      el.sessionView.classList.add("hidden");
      el.logoutBtn.classList.add("hidden");
      return;
    }
    refreshDashboardData();
    renderAchievements();
    renderLibrary();
    showDashboard();
  }

  function getCurrentScenario() {
    if (!state.session) {
      return null;
    }
    return getScenarioById(state.session.queue[state.session.currentIndex]);
  }

  function getScenarioById(id) {
    return SCENARIOS.find((s) => s.id === id) || null;
  }

  function clearTimer() {
    if (timerHandle) {
      window.clearInterval(timerHandle);
      timerHandle = null;
    }
  }

  function clamp(value, min, max) {
    return Math.min(max, Math.max(min, value));
  }

  function shuffle(values) {
    const arr = values.slice();
    for (let i = arr.length - 1; i > 0; i -= 1) {
      const j = Math.floor(Math.random() * (i + 1));
      const temp = arr[i];
      arr[i] = arr[j];
      arr[j] = temp;
    }
    return arr;
  }

  function escapeHtml(value) {
    return String(value)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#39;");
  }

  async function bootstrap() {
    apiAvailable = await checkApiHealth();
    if (apiAvailable) {
      const userId = getUserId();
      if (userId) {
        const serverState = await loadStateFromServer();
        if (serverState) {
          state = loadImportedState(serverState);
        }
      }
    }
    syncUI();
  }

  function getUserId() {
    return localStorage.getItem(USER_ID_KEY);
  }

  function setUserId(id) {
    localStorage.setItem(USER_ID_KEY, id);
  }

  function clearUserId() {
    localStorage.removeItem(USER_ID_KEY);
  }

  async function checkApiHealth() {
    try {
      const result = await apiRequest("/api/health", { method: "GET" });
      return Boolean(result && result.ok);
    } catch (error) {
      return false;
    }
  }

  async function apiLogin(name) {
    try {
      return await apiRequest("/api/login", {
        method: "POST",
        body: JSON.stringify({ name })
      });
    } catch (error) {
      return null;
    }
  }

  async function loadStateFromServer() {
    const userId = getUserId();
    if (!userId) {
      return null;
    }
    try {
      return await apiRequest("/api/state", {
        method: "GET",
        headers: {
          "x-user-id": userId
        }
      });
    } catch (error) {
      return null;
    }
  }

  async function saveStateToServer() {
    const userId = getUserId();
    if (!userId) {
      return;
    }
    try {
      await apiRequest("/api/state", {
        method: "POST",
        headers: {
          "x-user-id": userId
        },
        body: JSON.stringify(state)
      });
    } catch (error) {
      return;
    }
  }

  async function apiRequest(path, options) {
    const controller = new AbortController();
    const timeout = window.setTimeout(() => controller.abort(), API_TIMEOUT_MS);
    try {
      const response = await fetch(path, {
        ...options,
        headers: {
          "Content-Type": "application/json",
          ...(options && options.headers ? options.headers : {})
        },
        signal: controller.signal
      });
      if (!response.ok) {
        throw new Error(`Request failed: ${response.status}`);
      }
      return await response.json();
    } finally {
      window.clearTimeout(timeout);
    }
  }
})();

const CACHE_NAME = "aegisphish-cache-v1";
const ASSETS = [
  ".",
  "index.html",
  "app.css",
  "app.js",
  "manifest.json",
  "icon-180.png",
  "icon.svg",
  "icon-maskable.svg"
];

self.addEventListener("install", (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => cache.addAll(ASSETS))
  );
});

self.addEventListener("activate", (event) => {
  event.waitUntil(
    caches.keys().then((keys) =>
      Promise.all(keys.filter((key) => key !== CACHE_NAME).map((key) => caches.delete(key)))
    )
  );
});

self.addEventListener("fetch", (event) => {
  const { request } = event;
  if (request.method !== "GET") {
    return;
  }
  event.respondWith(
    caches.match(request).then((cached) => cached || fetch(request))
  );
});
